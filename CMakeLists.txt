cmake_minimum_required(VERSION 3.6)
project(external)

set(COCOS_EXTERNAL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")

set(_cocos_external_full_platform FALSE)
if(MACOSX OR ANDROID OR LINUX)
    set(_cocos_external_full_platform TRUE)
else()
    message(STATUS
        "Dependency source builds are not tested on ${CMAKE_SYSTEM_NAME}; "
        "attempting the available integrations.")
endif()

add_library(external empty.cpp)

include(cmake/zlib.cmake)
include(cmake/freetype.cmake)

include(cmake/box2d.cmake)
include(cmake/chipmunk.cmake)
include(cmake/fastlz.cmake)
include(cmake/recast.cmake)
include(cmake/bullet.cmake)
include(cmake/jpeg.cmake)
include(cmake/tiff.cmake)
include(cmake/uv.cmake)
include(cmake/webp.cmake)
include(cmake/tinyxml2.cmake)
include(cmake/tinydir.cmake)
include(cmake/xxhash.cmake)
include(cmake/xxtea.cmake)
include(cmake/clipper.cmake)
include(cmake/edtaa3func.cmake)
include(cmake/convertutf.cmake)
include(cmake/poly2tri.cmake)
include(cmake/md5.cmake)

target_link_libraries(external
    ext_zlib
    ext_box2d
    ext_chipmunk
    ext_freetype
    ext_recast
    ext_jpeg
    ext_uv
    ext_webp
    ext_bullet
    ext_tiff
    ext_tinyxml2
    ext_tinydir
    ext_xxhash
    ext_xxtea
    ext_clipper
    ext_edtaa3func
    ext_convertUTF
    ext_poly2tri
    ext_md5
)

if(_cocos_external_full_platform)
    include(cmake/openssl.cmake)
    include(cmake/websockets.cmake)
    target_link_libraries(external
        ext_ssl
        ext_crypto
        ext_websockets
    )
else()
    message(STATUS
        "Skipping OpenSSL and libwebsockets source integration on "
        "${CMAKE_SYSTEM_NAME}.")
endif()

# The engine includes jpeglib.h directly, so this public include directory must
# propagate through the aggregate external target instead of relying on a
# machine-wide JPEG installation.
target_include_directories(external INTERFACE
    "${COCOS_EXTERNAL_ROOT}"
    "${COCOS_EXTERNAL_ROOT}/include"
    "${COCOS_EXTERNAL_ROOT}/json/include"
    "${COCOS_EXTERNAL_ROOT}/jpeg"
    "${CMAKE_CURRENT_BINARY_DIR}/jpeg"
)

# use lua/js specific libs by property to prevent conflict
if(BUILD_JS_LIBS)
    if(NOT EXISTS "${COCOS_EXTERNAL_ROOT}/spidermonkey/CMakeLists.txt")
        message(FATAL_ERROR
            "BUILD_JS_LIBS requires SpiderMonkey, whose legacy downloaded "
            "binary bundle is no longer part of the source-only checkout.")
    endif()
    add_subdirectory(spidermonkey)
    set_property(TARGET external APPEND PROPERTY
        CC_JS_DEPEND ext_spidermonkey
    )
endif()
if(BUILD_LUA_LIBS AND (MACOSX OR ANDROID OR LINUX))
    include(cmake/luajit.cmake)
    include(cmake/tolua.cmake)
    include(cmake/luasocket.cmake)
    set_property(TARGET external APPEND PROPERTY
        CC_LUA_DEPEND ext_luajit ext_tolua ext_luasocket
    )
elseif(BUILD_LUA_LIBS)
    message(STATUS
        "Skipping LuaJIT source integration on ${CMAKE_SYSTEM_NAME}.")
endif()

include(cmake/png.cmake)
target_link_libraries(external
    ext_png
)

# png.h includes the configuration header generated by libpng's CMake
# build.  Propagate both locations to Cocos sources and consumers.
target_include_directories(external INTERFACE
    "${COCOS_EXTERNAL_ROOT}/png"
    "${CMAKE_CURRENT_BINARY_DIR}/png"
)

if(_cocos_external_full_platform)
    include(cmake/curl.cmake)
    target_link_libraries(external ext_curl)
endif()

if(ANDROID)
    add_subdirectory(android-specific/cpufeatures)
    add_subdirectory(android-specific/pvmp3dec)
    add_subdirectory(android-specific/tremolo)
    target_link_libraries(external
        ext_cpufeatures
        ext_pvmp3dec
        ext_tremolo
    )
endif()

if(MACOSX OR LINUX)
    include(cmake/glfw.cmake)
    target_link_libraries(external
        ext_glfw3
    )
endif()

if(LINUX)
    include(cmake/miniaudio.cmake)
    target_link_libraries(external ext_miniaudio)
endif()

# unzip depend on zlib
include(cmake/unzip.cmake)
target_link_libraries(external ext_unzip)

# put "external" into External folder, too
set_target_properties(external
    PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib"
    FOLDER "External"
)

unset(_cocos_external_full_platform)
